var SHEET_NAME = 'log'; var TZ = 'Asia/Tokyo'; var TABLE_ROWS = 60; var SCAN_ROWS = 4000; // 合言葉。必ず自分だけが分かる文字列に変えてください var RESET_KEY = 'change-me-1234'; function sheet_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = ss.getSheetByName(SHEET_NAME); if (!sh) { sh = ss.insertSheet(SHEET_NAME); sh.appendRow(['timestamp', 'temp_c']); } return sh; } function stats_() { var props = PropertiesService.getScriptProperties(); var raw = props.getProperty('stats'); if (raw) return JSON.parse(raw); return rebuildStats_(); } function saveStats_(s) { PropertiesService.getScriptProperties().setProperty('stats', JSON.stringify(s)); } function rebuildStats_() { var sh = sheet_(); var n = sh.getLastRow(); var s = { max: null, min: null, count: 0 }; if (n >= 2) { var vals = sh.getRange(2, 2, n - 1, 1).getValues(); for (var i = 0; i < vals.length; i++) { var t = Number(vals[i][0]); if (isNaN(t) || vals[i][0] === '') continue; s.count++; if (s.max === null || t > s.max) s.max = t; if (s.min === null || t < s.min) s.min = t; } } saveStats_(s); return s; } function doPost(e) { try { var body = JSON.parse(e.postData.contents); var rows = body.rows || []; if (rows.length) { var sh = sheet_(); var out = []; var s = stats_(); for (var i = 0; i < rows.length; i++) { var t = Number(rows[i][1]); out.push([new Date(rows[i][0]), t]); if (!isNaN(t)) { s.count++; if (s.max === null || t > s.max) s.max = t; if (s.min === null || t < s.min) s.min = t; } } sh.getRange(sh.getLastRow() + 1, 1, out.length, 2).setValues(out); saveStats_(s); CacheService.getScriptCache().remove('summary'); } return json_({ ok: true, added: rows.length }); } catch (err) { return json_({ ok: false, error: String(err) }); } } function doGet(e) { try { var p = (e && e.parameter) ? e.parameter : {}; if (p.reset) { if (p.reset !== RESET_KEY) return json_({ ok: false, error: 'key mismatch' }); if (p.mode === 'delete') { purgeLog_(); return json_({ ok: true, done: 'purged' }); } var name = archiveLog_(); return json_({ ok: true, done: 'archived', sheet: name }); } if (p.rebuild) { var s = rebuildStats_(); return json_({ ok: true, done: 'rebuilt', stats: s }); } return json_(summary_()); } catch (err) { return json_({ ok: false, error: String(err) }); } } function summary_() { var cache = CacheService.getScriptCache(); var hit = cache.get('summary'); if (hit) return JSON.parse(hit); var sh = sheet_(); var n = sh.getLastRow(); if (n < 2) return { ok: true, empty: true }; var start = Math.max(2, n - SCAN_ROWS + 1); var vals = sh.getRange(start, 1, n - start + 1, 2).getValues(); var last = null; var buckets = {}; for (var i = 0; i < vals.length; i++) { var d = vals[i][0]; var t = Number(vals[i][1]); if (!(d instanceof Date) || isNaN(t)) continue; last = { at: d, temp: t }; var key = Utilities.formatDate(d, TZ, 'yyyyMMddHHmm'); var b = buckets[key]; if (!b) { b = buckets[key] = { label: Utilities.formatDate(d, TZ, 'HH:mm'), min10: (Number(Utilities.formatDate(d, TZ, 'mm')) % 10 === 0), max: t, min: t, sum: 0, cnt: 0 }; } if (t > b.max) b.max = t; if (t < b.min) b.min = t; b.sum += t; b.cnt++; } if (!last) return { ok: true, empty: true }; var keys = Object.keys(buckets).sort().reverse().slice(0, TABLE_ROWS); var minutes = keys.map(function (k) { var b = buckets[k]; return { label: b.label, tenMin: b.min10, max: round1_(b.max), min: round1_(b.min), avg: round1_(b.sum / b.cnt), cnt: b.cnt }; }); var st = stats_(); var res = { ok: true, temp: round1_(last.temp), at: Utilities.formatDate(last.at, TZ, 'yyyy-MM-dd HH:mm:ss'), ageSec: Math.round((Date.now() - last.at.getTime()) / 1000), allMax: st.max === null ? round1_(last.temp) : round1_(st.max), allMin: st.min === null ? round1_(last.temp) : round1_(st.min), count: st.count, minutes: minutes }; cache.put('summary', JSON.stringify(res), 4); return res; } function round1_(v) { return Math.round(v * 10) / 10; } function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); } function onOpen() { SpreadsheetApp.getUi() .createMenu('温度ロガー') .addItem('ログを退避して新しく始める', 'menuArchive_') .addItem('ログを完全に削除', 'menuPurge_') .addToUi(); } function menuArchive_() { var ui = SpreadsheetApp.getUi(); if (ui.alert('確認', '今のログを別シートに退避して、新しく記録を始めます。よろしいですか？', ui.ButtonSet.YES_NO) !== ui.Button.YES) return; var name = archiveLog_(); ui.alert('完了', '「' + name + '」に退避しました。', ui.ButtonSet.OK); } function menuPurge_() { var ui = SpreadsheetApp.getUi(); if (ui.alert('確認', 'ログを完全に削除します。元に戻せません。よろしいですか？', ui.ButtonSet.YES_NO) !== ui.Button.YES) return; purgeLog_(); ui.alert('完了', '削除しました。', ui.ButtonSet.OK); } function archiveLog_() { var ss = SpreadsheetApp.getActiveSpreadsheet(); var sh = sheet_(); var name = 'log_' + Utilities.formatDate(new Date(), TZ, 'yyyyMMdd_HHmm'); sh.setName(name); var fresh = ss.insertSheet(SHEET_NAME); fresh.appendRow(['timestamp', 'temp_c']); clearAll_(); return name; } function purgeLog_() { var sh = sheet_(); var n = sh.getLastRow(); if (n > 1) sh.getRange(2, 1, n - 1, sh.getLastColumn()).clearContent(); clearAll_(); } function clearAll_() { PropertiesService.getScriptProperties().deleteProperty('stats'); CacheService.getScriptCache().remove('summary'); }
